FIRESTORE SECURITY RULES - PRODUCTION READY
==========================================

Copy these rules to Firebase Console → Firestore Database → Rules:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ===================================================================
    // HELPER FUNCTIONS
    // ===================================================================

    // Check if user is authenticated
    function isSignedIn() {
      return request.auth != null;
    }

    // Check if user is admin (requires custom claim 'admin: true')
    // Set via Firebase Admin SDK: admin.auth().setCustomUserClaims(uid, { admin: true })
    function isAdmin() {
      return isSignedIn() && request.auth.token.admin == true;
    }

    // Check if user owns this customer record
    function isCustomerOwner(customerId) {
      return isSignedIn() && request.auth.uid == customerId;
    }

    // Check if email matches the customer's email in the document
    function isCustomerByEmail() {
      return isSignedIn() && request.auth.token.email == resource.data.email;
    }

    // Check if the order belongs to the authenticated user's email
    function isOrderOwner() {
      return isSignedIn() &&
             request.auth.token.email == resource.data.customer.email;
    }

    // Validate order creation data
    function validOrderCreate() {
      let data = request.resource.data;
      return data.keys().hasAll(['orderNumber', 'status', 'customer',
                                  'shippingAddress', 'items', 'total',
                                  'createdAt', 'updatedAt']) &&
             data.status == 'pending' &&
             data.items.size() > 0 &&
             data.total >= 0 &&
             data.customer.email is string &&
             data.customer.email.matches('.*@.*\\..*');
    }

    // Only allow specific fields to be updated
    function validOrderUpdate() {
      let affectedKeys = request.resource.data.diff(resource.data).affectedKeys();
      // Allow updating these fields only:
      // - paymentStatus, paidAt (for payment completion)
      // - status, statusHistory, tracking, notes (for admin updates)
      // - updatedAt (always allowed for tracking changes)
      let allowedFields = ['paymentStatus', 'paidAt', 'status', 'statusHistory',
                           'tracking', 'notes', 'updatedAt'].toSet();
      return affectedKeys.hasOnly(allowedFields);
    }

    // ===================================================================
    // PRODUCTS COLLECTION
    // ===================================================================

    match /products/{productId} {
      // Anyone can read products (public catalog)
      allow read: if true;

      // Only admins can create, update, or delete products
      allow create, update, delete: if isAdmin();
    }

    // ===================================================================
    // VIDEOS COLLECTION
    // ===================================================================

    match /videos/{videoId} {
      // Anyone can read videos (public resources)
      allow read: if true;

      // Only admins can manage videos
      allow create, update, delete: if isAdmin();
    }

    // ===================================================================
    // ORDERS COLLECTION
    // ===================================================================

    match /orders/{orderId} {
      // Read: Customer can read their own orders, admins can read all
      allow read: if isAdmin() ||
                     (resource.data.customer.email == request.auth.token.email);

      // Create: Anyone can create orders during checkout
      // But must include required fields and valid data
      allow create: if validOrderCreate();

      // Update: Admins can update anything, customers can only update payment status
      allow update: if isAdmin() ||
                       (isOrderOwner() && validOrderUpdate());

      // Delete: Only admins can delete orders
      allow delete: if isAdmin();
    }

    // ===================================================================
    // CUSTOMERS COLLECTION
    // ===================================================================

    match /customers/{customerId} {
      // Read: Customer can read their own data, admins can read all
      allow read: if isAdmin() || isCustomerOwner(customerId);

      // Create: Anyone can create during checkout (but limited fields)
      allow create: if request.resource.data.email is string &&
                       request.resource.data.email.matches('.*@.*\\..*');

      // Update: Only admins or the customer themselves
      allow update: if isAdmin() || isCustomerOwner(customerId);

      // Delete: Only admins can delete customers
      allow delete: if isAdmin();
    }

    // ===================================================================
    // MAIL COLLECTION (for Firebase Trigger Email Extension)
    // ===================================================================

    match /mail/{mailId} {
      // No one can read mail documents (contains email content)
      allow read: if false;

      // Only authenticated users (admins or system) can create emails
      // This prevents spam/abuse while allowing order emails to be sent
      allow create: if isSignedIn() || isAdmin();

      // Only the extension can update delivery status
      allow update: if false;

      // Only admins can delete (for cleanup)
      allow delete: if isAdmin();
    }

    // ===================================================================
    // MAIL_FAILURES COLLECTION (for failed emails)
    // ===================================================================

    match /mail_failures/{failureId} {
      // Only admins can read failed email logs
      allow read: if isAdmin();

      // No one can write (extension handles this)
      allow write: if false;
    }
  }
}

/*
To set admin privileges for a user, use Firebase Admin SDK or Firebase CLI:

METHOD 1: Firebase Admin SDK (Node.js)
---------------------------------------
const admin = require('firebase-admin');
admin.initializeApp();

// Set admin claim for a user
admin.auth().setCustomUserClaims('USER_UID_HERE', { admin: true })
  .then(() => console.log('Admin claim set'))
  .catch(error => console.error('Error:', error));

METHOD 2: Firebase CLI
----------------------
1. Install Firebase tools: npm install -g firebase-tools
2. Login: firebase login
3. Create a script to set claims (see ADMIN_SETUP.md)

METHOD 3: Cloud Function (Recommended)
---------------------------------------
Create a callable Cloud Function that admins can use to promote users.
See ADMIN_SETUP.md for complete implementation.

IMPORTANT: After setting custom claims, users must sign out and sign in again
for the new claims to take effect.
*/
